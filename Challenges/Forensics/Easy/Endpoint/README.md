> - Tạo file `extract.py` truyền vào file chứa bản ghi các truy vấn chèn dữ liệu chương trình trích suất dữ liệu được truyền vào và giải mã base64 sau đó ghi vào file khác 

```python
#!/usr/bin/env python3
import re
import base64
import sys
from pathlib import Path

print(f"Usage: {sys.argv[0]} dump.sql output.so")

sql_file = Path(sys.argv[1])
output_file = Path(sys.argv[2])

sql = sql_file.read_text(encoding="utf-8", errors="replace")
print(f"[+] Read file: {len(sql)}")

pattern = (
    r"INSERT\s+INTO\s+`?xQGgYA`?\s+"
    r"VALUES\s*\(\s*'([^']*)'\s*\)"
)
chunks = re.findall(pattern, sql, flags=re.IGNORECASE)
print(f"[+] Extract data: {len(chunks)}")

encoded = "".join(chunks)

binary = base64.b64decode(encoded, validate=True)
print("[+] Decode Base64")

output_file.write_bytes(binary)
print(f"[+] Write File: {output_file}")
```
